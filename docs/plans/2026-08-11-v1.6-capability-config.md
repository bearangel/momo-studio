# v1.6 Agent 能力配置 + Marketplace 自定义上传 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 修复 agent 能力配置三处断裂（merged.tools 丢失 bug + UI 无编辑器 + Marketplace 无自定义入口），让 v1.5 全部 24 工具的允许列表真正生效；同时新增 Layer 3 per-assignment override、自定义 MCP 注册、Skill zip 上传三类用户能力配置入口。

**Architecture:** 三层能力合并模型（Layer 1 def + Layer 2 workspace + Layer 3 assignment delta）。Migration v16 加 `agent_assignment_capabilities` 表 + `mcp_definitions.source/installed_at` 列 + builtin default_tools 修复。修复 `spawn-helpers.buildSpawnOpts` 把 `merged.tools` 注入 `allowedTools`。Renderer 加共享 `CapabilityTabs` 组件被 DefinitionEditor 和 Layer 3 弹窗复用。Marketplace 加自定义 MCP/Skill 入口。

**Tech Stack:** TypeScript strict / Node 20 LTS / Electron / better-sqlite3 / React / vitest / js-yaml / adm-zip

**Spec 来源：** `docs/specs/2026-08-11-v1.6-capability-config-design.md`

## Global Constraints

- **Node 版本**：`nvm use 20` 后 `export PATH="/home/ai-agent/.nvm/versions/node/v20.20.2/bin:$PATH"`，禁止用 Node 26（better-sqlite3 binding 失效）
- **包管理器**：`npx pnpm@9.0.0`，单 workspace 命令加 `--filter momo-studio-electron` 或 `--filter momo-studio-renderer`
- **TypeScript strict**：禁止 `any` / `@ts-ignore` / `as any`，禁止 `// eslint-disable`
- **测试命令**：`cd electron && npx pnpm@9.0.0 vitest run tests/<path>` 或 `cd renderer && npx pnpm@9.0.0 vitest run`
- **Typecheck 命令**：根目录 `npx pnpm@9.0.0 typecheck`（两个 workspace 同时）
- **中文注释**：所有源码注释使用中文（AGENTS.md 强制）
- **Conventional Commits**：`feat:` / `fix:` / `chore:` / `refactor:` / `test:` / `docs:`
- **Git 推送方式**：`git push https://x-access-token:GH_TOKEN@github.com/bearangel/momo-studio.git main`
- **docs/ 在 .gitignore**：用 `git add -f docs/your-file.md`
- **matrix-js-sdk 锁定 ^31.0.0**：禁止升级（CJS 主进程兼容）

```typescript
{
  version: 16,
  name: 'v1.6 能力配置：agent_assignment_capabilities 表 + mcp_definitions.source/installed_at 列 + builtin default_tools 修复',
  sql: [
    // 新表：per-assignment 能力 delta
    `CREATE TABLE IF NOT EXISTS agent_assignment_capabilities (
       assignment_id   TEXT NOT NULL REFERENCES agent_assignments(instance_id) ON DELETE CASCADE,
       capability_type TEXT NOT NULL CHECK (capability_type IN ('tool','mcp','skill')),
       mode            TEXT NOT NULL CHECK (mode IN ('add','remove')),
       ref             TEXT NOT NULL,
       PRIMARY KEY (assignment_id, capability_type, mode, ref)
     )`,
    // mcp_definitions 加 source + installed_at 列（IF NOT EXISTS 通过 PRAGMA 检查兼容）
    // SQLite 不支持 ADD COLUMN IF NOT EXISTS，需先查表信息（用 application_id 防御）
    // 简化：直接 ALTER，重复执行会报错——但 migration 一次性执行保证幂等
    `ALTER TABLE mcp_definitions ADD COLUMN source TEXT NOT NULL DEFAULT 'marketplace'`,
    `ALTER TABLE mcp_definitions ADD COLUMN installed_at TEXT NOT NULL DEFAULT (datetime('now'))`,
    // builtin default_tools 修复：写为 24 工具 JSON（保持 v1.5 行为不变）
    // JSON 数组结构：[{ kind: 'builtin', ref: 'tool_name' }, ...]
    `UPDATE agent_definitions
     SET default_tools = '${JSON.stringify(
       ['read_file','write_file','list_files','edit_file','mkdir','rm','mv','exists',
        'grep','glob','bash',
        'git_status','git_diff','git_log','git_show','git_add','git_commit','git_branch','git_checkout','git_stash',
        'webfetch','todowrite','lsp_diagnostics','lsp_find_references']
         .map((ref) => ({ kind: 'builtin', ref }))
     )}'
     WHERE source = 'builtin'`,
  ],
},

// electron/src/main/agent/assignment-capabilities.ts
//
// Per-assignment 能力 delta 持久化（Layer 3）。
// 数据表 agent_assignment_capabilities（migration v16 创建）。
// 合并算法在 capability-merger.ts 实现；本文件只管 CRUD。

import { getDb } from '../storage/db';
import type { CapabilityType } from '../workspace/allocation';

/** 某 assignment 的能力 delta（按 add/remove 双向） */
export interface AssignmentDeltas {
  addedTools: string[];   removedTools: string[];
  addedMcps: string[];    removedMcps: string[];
  addedSkills: string[];  removedSkills: string[];
}

interface DeltaRow {
  capability_type: string;
  mode: string;
  ref: string;
}

const EMPTY_DELTAS: AssignmentDeltas = {
  addedTools: [], removedTools: [],
  addedMcps: [], removedMcps: [],
  addedSkills: [], removedSkills: [],
};

/** 读取某 assignment 的 deltas。无 delta 时返回全空对象。 */
export function getAssignmentDeltas(assignmentId: string): AssignmentDeltas {
  const db = getDb();
  const rows = db
    .prepare(
      'SELECT capability_type, mode, ref FROM agent_assignment_capabilities WHERE assignment_id = ?',
    )
    .all(assignmentId) as DeltaRow[];

  const result: AssignmentDeltas = {
    addedTools: [], removedTools: [],
    addedMcps: [], removedMcps: [],
    addedSkills: [], removedSkills: [],
  };

  const buckets: Record<string, { add: string[]; remove: string[] }> = {
    tool: { add: result.addedTools, remove: result.removedTools },
    mcp: { add: result.addedMcps, remove: result.removedMcps },
    skill: { add: result.addedSkills, remove: result.removedSkills },
  };

  for (const row of rows) {
    const bucket = buckets[row.capability_type];
    if (!bucket) continue;
    if (row.mode === 'add') bucket.add.push(row.ref);
    else if (row.mode === 'remove') bucket.remove.push(row.ref);
  }
  return result;
}

/**
 * 全量替换某 assignment 的 deltas。
 * 内部：DELETE WHERE assignment_id = ? → INSERT 新值（事务）。
 * 幂等：同 deltas 多次保存结果一致。
 */
export function setAssignmentDeltas(assignmentId: string, deltas: AssignmentDeltas): void {
  const db = getDb();
  const insert = db.prepare(
    'INSERT INTO agent_assignment_capabilities (assignment_id, capability_type, mode, ref) VALUES (?, ?, ?, ?)',
  );
  const del = db.prepare('DELETE FROM agent_assignment_capabilities WHERE assignment_id = ?');

  const ops: Array<[string, string, string, string]> = [
    ...deltas.addedTools.map((r): [string, string, string, string] => [assignmentId, 'tool', 'add', r]),
    ...deltas.removedTools.map((r): [string, string, string, string] => [assignmentId, 'tool', 'remove', r]),
    ...deltas.addedMcps.map((r): [string, string, string, string] => [assignmentId, 'mcp', 'add', r]),
    ...deltas.removedMcps.map((r): [string, string, string, string] => [assignmentId, 'mcp', 'remove', r]),
    ...deltas.addedSkills.map((r): [string, string, string, string] => [assignmentId, 'skill', 'add', r]),
    ...deltas.removedSkills.map((r): [string, string, string, string] => [assignmentId, 'skill', 'remove', r]),
  ];

  const tx = db.transaction(() => {
    del.run(assignmentId);
    for (const op of ops) insert.run(...op);
  });
  tx();
}

// electron/src/main/agent/capability-merger.ts
//
// 三层能力叠加合并 —— 把 agent 定义默认能力 + workspace 级共享能力 +
// per-assignment delta 合并为最终可用清单。
//
// 三层模型：
//   Layer 1: AgentDefinition.default*   —— agent 定义自带的默认能力（base）
//   Layer 2: WorkspaceAllocation        —— workspace 内所有 agent 共享的能力（增量）
//   Layer 3: AssignmentDeltas           —— per-assignment 加/减 delta（v1.6 新增）
//
// 合并规则：
//   Step 1：Layer 1 ∪ Layer 2（并集去重）
//   Step 2：result = (Step 1 ∪ added) - removed
//   冲突：同 ref 同时 added + removed 时 removed 胜出（保守语义）

import type { AgentDefinition, ToolRef, McpRef, SkillRef } from './types';
import type { WorkspaceAllocation } from '../workspace/allocation';
import type { AssignmentDeltas } from './assignment-capabilities';

/** 合并后的最终可用能力清单（ref 字符串列表，已去重） */
export interface MergedCapabilities {
  tools: string[];
  mcps: string[];
  skills: string[];
}

function subtract(set: string[], removed: string[]): string[] {
  const rset = new Set(removed);
  return set.filter((x) => !rset.has(x));
}

function union(a: string[], b: string[]): string[] {
  return [...new Set([...a, ...b])];
}

/**
 * 三层能力叠加：default(def) ∪ workspace(allocation) ∪ added - removed。
 * 不传 deltas 时（v1.5 行为）仅合并 Layer 1 + Layer 2，向后兼容。
 */
export function mergeCapabilities(
  def: AgentDefinition,
  allocation: WorkspaceAllocation,
  deltas?: AssignmentDeltas,
): MergedCapabilities {
  const defTools = def.defaultTools.map((t: ToolRef) => t.ref);
  const defMcps = def.defaultMcps.map((m: McpRef) => m.ref);
  const defSkills = def.defaultSkills.map((s: SkillRef) => s.ref);

  // Step 1：Layer 1 ∪ Layer 2
  let tools = [...new Set([...defTools, ...allocation.tools])];
  let mcps = [...new Set([...defMcps, ...allocation.mcps])];
  let skills = [...new Set([...defSkills, ...allocation.skills])];

  // Step 2：apply Layer 3 deltas（仅当传入时）
  if (deltas) {
    tools = subtract(union(tools, deltas.addedTools), deltas.removedTools);
    mcps = subtract(union(mcps, deltas.addedMcps), deltas.removedMcps);
    skills = subtract(union(skills, deltas.addedSkills), deltas.removedSkills);
  }

  return { tools, mcps, skills };
}
```

/** 列出所有已注册 MCP（含 source 区分） */
export function listRegistered(): Required<McpServerConfig>[] {
  const db = getDb();
  const rows = db.prepare(
    'SELECT id, name, version, command, args, env, source, installed_at FROM mcp_definitions ORDER BY installed_at DESC'
  ).all() as Array<{
    id: string; name: string; version: string; command: string;
    args: string; env: string | null; source: string; installed_at: string;
  }>;
  return rows.map((r) => ({
    id: r.id, name: r.name, version: r.version, command: r.command,
    args: JSON.parse(r.args) as string[],
    env: r.env ? JSON.parse(r.env) as Record<string, string> : undefined,
    source: r.source as 'marketplace' | 'custom',
    installedAt: r.installed_at,
  }));
}

/** 删除已注册 MCP；marketplace 装的不可删（提示用户用卸载按钮） */
export function deleteRegistered(name: string): void {
  const db = getDb();
  const row = db.prepare('SELECT source FROM mcp_definitions WHERE name = ?').get(name) as { source: string } | undefined;
  if (!row) return;  // 不存在静默跳过
  if (row.source === 'marketplace') {
    throw new Error(`MCP ${name} 是 marketplace 装的，请用卸载按钮移除`);
  }
  db.prepare('DELETE FROM mcp_definitions WHERE name = ?').run(name);
}

// electron/src/main/skill/zip-uploader.ts
//
// 自定义 Skill zip 上传：解压 + SHA256 校验 + 注册到 <userData>/skills/<slug>。
// zip 结构：<slug>/SKILL.md + 可选 <slug>/resources/*。
// 同 slug 同 SHA256 二次上传幂等（不重写）；同 slug 不同 SHA256 覆盖。

import AdmZip from 'adm-zip';
import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import { resolveSkillsDir } from '../paths';
import { logger } from '../logger';

export interface InstalledSkill {
  slug: string;
  name: string;
  description: string;
  source: 'builtin' | 'marketplace' | 'custom';
  installedAt: string | null;
}

export function getSkillsDir(): string {
  return resolveSkillsDir();
}

interface SkillFrontmatter {
  slug?: string;
  name?: string;
  description?: string;
}

function parseFrontmatter(md: string): SkillFrontmatter {
  const m = md.match(/^---\n([\s\S]*?)\n---/);
  if (!m) return {};
  const front: SkillFrontmatter = {};
  for (const line of m[1].split('\n')) {
    const mm = line.match(/^(\w+):\s*(.*)$/);
    if (mm) (front as Record<string, string>)[mm[1]] = mm[2];
  }
  return front;
}

/** 上传 zip → 解压 + 注册。返回 slug + description。 */
export function uploadSkillZip(buffer: Buffer, _filename: string): { slug: string; description: string } {
  const zip = new AdmZip(buffer);
  const entries = zip.getEntries();
  // 找 SKILL.md 文件（路径形如 <slug>/SKILL.md）
  const skillEntries = entries.filter((e) => !e.isDirectory && e.entryName.endsWith('/SKILL.md') || e.entryName === 'SKILL.md');
  if (skillEntries.length === 0) {
    throw new Error('zip 内未找到 SKILL.md');
  }
  if (skillEntries.length > 1) {
    throw new Error('zip 根目录包含多个 SKILL.md（应有且仅有一个）');
  }
  const skillEntry = skillEntries[0];
  const parts = skillEntry.entryName.split('/');
  // 期望：<slug>/SKILL.md（或 SKILL.md）
  if (parts.length > 2) {
    throw new Error(`SKILL.md 路径过深：${skillEntry.entryName}`);
  }
  const slug = parts.length === 2 ? parts[0] : 'unnamed';

  // 解析 frontmatter
  const md = skillEntry.getData().toString('utf-8');
  const front = parseFrontmatter(md);
  const description = front.description ?? '';

  // SHA256 幂等检查
  const hash = crypto.createHash('sha256').update(buffer).digest('hex');
  const skillsDir = getSkillsDir();
  const targetDir = path.join(skillsDir, slug);
  const hashFile = path.join(targetDir, '.sha256');
  if (fs.existsSync(hashFile) && fs.readFileSync(hashFile, 'utf-8') === hash) {
    logger.info('Skill zip 同 hash 幂等返回', { slug, hash });
    return { slug, description };
  }

  // 备份原目录（如有）
  if (fs.existsSync(targetDir)) {
    const bak = `${targetDir}.bak.${Date.now()}`;
    fs.renameSync(targetDir, bak);
    // 异步清理 .bak（简单起见同步删除）
    fs.rmSync(bak, { recursive: true, force: true });
  }

  // 解压到目标目录
  fs.mkdirSync(targetDir, { recursive: true });
  for (const entry of entries) {
    if (entry.isDirectory) continue;
    const rel = entry.entryName.startsWith(`${slug}/`)
      ? entry.entryName.slice(slug.length + 1)
      : entry.entryName;
    if (!rel || rel.includes('..')) continue;  // 路径防御
    const dest = path.join(targetDir, rel);
    fs.mkdirSync(path.dirname(dest), { recursive: true });
    fs.writeFileSync(dest, entry.getData());
  }
  // 写 hash 文件
  fs.writeFileSync(hashFile, hash);
  logger.info('Skill zip 上传成功', { slug, hash });
  return { slug, description };
}

```typescript
// renderer/src/components/agent/CapabilityTabs.tsx
//
// 能力配置共享组件：三个 Tab（工具/MCP/Skill）+ 类别分组 checkbox。
// 三种模式：
//   - mode='edit'：value={ tools:[], mcps:[], skills:[] }（绝对勾选集合）
//   - mode='override'：value={ tools, mcps, skills }（最终集合，对照 defaultValue 计算 delta）
//   - mode='readonly'：checkbox disabled
// 底部三个快捷按钮：[全选] [清空] [安全最小集]（仅 edit 模式工具 Tab 显示）
//
// 被 DefinitionEditor（edit / readonly 模式）和 AssignmentCapabilitiesDialog（override 模式）复用。

import { useEffect, useState } from 'react';
import { ipc } from '../../ipc/client';
import { cn } from '../../lib/cn';
import {
  ALL_BUILTIN_TOOLS, SAFE_MINIMUM_TOOLS, TOOL_CATEGORIES,
} from '../../lib/tool-catalog';
import type { RegisteredMcp, InstalledSkill } from '../../ipc/types';

type Tab = 'tools' | 'mcp' | 'skill';

interface Capabilities {
  tools: string[];
  mcps: string[];
  skills: string[];
}

interface Props {
  mode: 'edit' | 'override' | 'readonly';
  /** def+ws 默认集合（override 模式下用于计算 delta；其他模式仅用于显示提示） */
  defaultValue?: Capabilities;
  /** 当前选中集合（edit 模式：绝对值；override 模式：最终值） */
  value: Capabilities;
  onChange: (next: Capabilities) => void;
}

export function CapabilityTabs({ mode, defaultValue, value, onChange }: Props) {
  const [tab, setTab] = useState<Tab>('tools');
  const [mcps, setMcps] = useState<RegisteredMcp[]>([]);
  const [skills, setSkills] = useState<InstalledSkill[]>([]);

  useEffect(() => {
    void ipc.mcp.listRegistered().then(setMcps);
    void ipc.skill.listInstalled().then(setSkills);
  }, []);

  const readonly = mode === 'readonly';

  return (
    <div className="flex flex-col gap-2">
      <div className="flex gap-1 border-b border-border-subtle">
        {(['tools', 'mcp', 'skill'] as const).map((t) => (
          <button
            key={t}
            type="button"
            className={cn(
              'text-xs px-2.5 py-1 -mb-px border-b-2',
              tab === t ? 'border-accent-blue text-accent-blue' : 'border-transparent text-neutral-400',
            )}
            onClick={() => setTab(t)}
          >
            {t === 'tools' ? '工具' : t === 'mcp' ? 'MCP' : 'Skill'}
          </button>
        ))}
      </div>

      {tab === 'tools' && (
        <div className="flex flex-col gap-2">
          {TOOL_CATEGORIES.map((cat) => (
            <div key={cat.label}>
              <div className="text-xs text-neutral-500 mb-1">{cat.emoji} {cat.label}</div>
              <div className="flex flex-wrap gap-2">
                {cat.tools.map((tool) => {
                  const checked = value.tools.includes(tool);
                  return (
                    <label key={tool} className="flex items-center gap-1 text-xs text-neutral-300">
                      <input
                        type="checkbox"
                        aria-label={tool}
                        checked={checked}
                        disabled={readonly}
                        onChange={(e) => {
                          if (mode === 'readonly') return;
                          const next = e.target.checked
                            ? [...value.tools, tool]
                            : value.tools.filter((t) => t !== tool);
                          onChange({ ...value, tools: next });
                        }}
                      />
                      {tool}
                    </label>
                  );
                })}
              </div>
            </div>
          ))}

          {mode === 'edit' && (
            <div className="flex gap-1 mt-1">
              <button type="button" className="text-xs px-2 py-0.5 rounded bg-bg-tertiary hover:bg-border-subtle"
                onClick={() => onChange({ ...value, tools: [...ALL_BUILTIN_TOOLS] })}>全选</button>
              <button type="button" className="text-xs px-2 py-0.5 rounded bg-bg-tertiary hover:bg-border-subtle"
                onClick={() => onChange({ ...value, tools: [] })}>清空</button>
              <button type="button" className="text-xs px-2 py-0.5 rounded bg-bg-tertiary hover:bg-border-subtle"
                onClick={() => onChange({ ...value, tools: [...SAFE_MINIMUM_TOOLS] })}>安全最小集</button>
            </div>
          )}

          {mode === 'override' && defaultValue && (
            <div className="text-xs text-neutral-500">
              默认（def + workspace）：{defaultValue.tools.join(', ') || '无'}
              <br />
              勾选 = 保留/添加；取消 = 移除
            </div>
          )}
        </div>
      )}

      {tab === 'mcp' && (
        <div className="flex flex-col gap-1">
          {mcps.length === 0 ? (
            <div className="text-xs text-neutral-500">尚未注册任何 MCP（去 Marketplace → + 添加 MCP）</div>
          ) : mcps.map((m) => {
            const checked = value.mcps.includes(m.name);
            return (
              <label key={m.name} className="flex items-center gap-1 text-xs text-neutral-300">
                <input
                  type="checkbox"
                  aria-label={m.name}
                  checked={checked}
                  disabled={readonly}
                  onChange={(e) => {
                    if (mode === 'readonly') return;
                    const next = e.target.checked
                      ? [...value.mcps, m.name]
                      : value.mcps.filter((x) => x !== m.name);
                    onChange({ ...value, mcps: next });
                  }}
                />
                {m.name}
                <span className="text-neutral-500">[{m.source}]</span>
              </label>
            );
          })}
        </div>
      )}

      {tab === 'skill' && (
        <div className="flex flex-col gap-1">
          {skills.length === 0 ? (
            <div className="text-xs text-neutral-500">尚未安装任何 Skill</div>
          ) : skills.map((s) => {
            const checked = value.skills.includes(s.slug);
            return (
              <label key={s.slug} className="flex items-center gap-1 text-xs text-neutral-300">
                <input
                  type="checkbox"
                  aria-label={s.slug}
                  checked={checked}
                  disabled={readonly}
                  onChange={(e) => {
                    if (mode === 'readonly') return;
                    const next = e.target.checked
                      ? [...value.skills, s.slug]
                      : value.skills.filter((x) => x !== s.slug);
                    onChange({ ...value, skills: next });
                  }}
                />
                {s.name} ({s.slug}) <span className="text-neutral-500">[{s.source}]</span>
              </label>
            );
          })}
        </div>
      )}
    </div>
  );
}
```
