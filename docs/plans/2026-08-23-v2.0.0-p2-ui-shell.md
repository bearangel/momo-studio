```ts
// electron/src/main/window-state.ts
//
// 窗口状态持久化（bounds + maximized）——kv_store key='window_state'。
// 独立于 electron 模块（纯 DB 读写），便于单测。
import { getDb } from './storage/db';

export interface WindowState { x: number; y: number; width: number; height: number; maximized: boolean; }
const KEY = 'window_state';

export function saveWindowState(s: WindowState): void {
  getDb().prepare(
    `INSERT INTO kv_store (key, value, updated_at) VALUES (?, ?, datetime('now'))
     ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = excluded.updated_at`,
  ).run(KEY, JSON.stringify(s));
}

export function loadWindowState(): WindowState | null {
  const row = getDb().prepare('SELECT value FROM kv_store WHERE key = ?').get(KEY) as { value: string } | undefined;
  if (!row) return null;
  const parsed = JSON.parse(row.value) as Partial<WindowState>;
  if (typeof parsed.width !== 'number' || typeof parsed.height !== 'number') return null;
  return {
    x: typeof parsed.x === 'number' ? parsed.x : undefined as unknown as number, // 由调用方 bounds 校验兜底
    y: typeof parsed.y === 'number' ? parsed.y : undefined as unknown as number,
    width: parsed.width, height: parsed.height,
    maximized: parsed.maximized === true,
  };
}
```

（`undefined as unknown as number` 不合规——实际实现：返回 `{ x: number | null; ... }` 形状或直接过滤非法；以 typecheck clean 为准，上面仅为骨架意图，实现者按 strict 修正为可空字段 + 调用方判空。）

```ts
// electron/src/main/window-ipc.ts
// 窗口控制 IPC——renderer 的自绘 titlebar 控件调用。
import { ipcMain, BrowserWindow } from 'electron';

export function registerWindowIpc(getWin: () => BrowserWindow | null): void {
  ipcMain.on('window:minimize', () => getWin()?.minimize());
  ipcMain.on('window:toggle-maximize', () => {
    const win = getWin();
    if (!win) return;
    if (win.isMaximized()) win.unmaximize(); else win.maximize();
  });
  ipcMain.on('window:close', () => getWin()?.close());
  ipcMain.handle('window:is-maximized', () => getWin()?.isMaximized() ?? false);
  logger.info('Window IPC 已注册');
}
```

```tsx
// renderer/src/components/layout/TitleBar.tsx
// 顶部标题栏：Logo + workspace tabs + 拖拽区 + 窗口控件（win/linux 自绘；mac 留红绿灯空间）。
// -webkit-app-region: drag 使整条可拖动；交互元素（tab/按钮）标 no-drag。
import { useEffect, useState } from 'react';
import { isMac } from '../../lib/platform'; // process.platform 经 preload 注入（见下）
import { WorkspaceTabs } from './WorkspaceTabs';
import { ipc } from '../../ipc/client';

export function TitleBar() {
  const [maximized, setMaximized] = useState(false);
  useEffect(() => {
    void ipc.window.isMaximized().then(setMaximized);
    return ipc.window.onMaximizedChanged(setMaximized);
  }, []);
  return (
    <header
      className="h-10 shrink-0 flex items-center gap-1 bg-bg-secondary border-b border-border-subtle px-2 select-none"
      style={{ WebkitAppRegion: 'drag' } as React.CSSProperties}
    >
      {!isMac && <span className="pl-1 pr-2 text-xs text-neutral-400">◉ Momo Studio</span>}
      <WorkspaceTabs />
      <div className="flex-1 h-full" />
      {!isMac && (
        <div className="flex h-full" style={{ WebkitAppRegion: 'no-drag' } as React.CSSProperties}>
          <WinCtlBtn label="─" onClick={() => ipc.window.minimize()} />
          <WinCtlBtn label={maximized ? '❐' : '▢'} onClick={() => ipc.window.toggleMaximize()} />
          <WinCtlBtn label="✕" danger onClick={() => ipc.window.close()} />
        </div>
      )}
    </header>
  );
}
```
